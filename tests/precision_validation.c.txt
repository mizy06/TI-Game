#include "../waveform_analyzer.h"

/*
 * 版本 2 高精度接口回归测试。
 *
 * 与 host_test.c.txt 的默认参数测试不同，本文件会改变采样率、参考电压、
 * 前端增益、方波边沿宽度和噪声带宽，并直接调用 waveform_analyze_frame()。
 * 它验证算法确实使用每帧元数据换算物理量，而不是暗中写死 20 kHz/3.3 V。
 */

#include <math.h>
#include <stdint.h>
#include <stdio.h>

#ifndef M_PI
#define M_PI 3.14159265358979323846
#endif

#define CASES_PER_CLASS 600U

/* 固定种子保证 4*600 个强干扰用例可重复。 */
static uint32_t g_rng = 0x6D2B79F5U;
static double g_noise[WAVEFORM_SAMPLE_COUNT];

/* 与产品算法无关的测试随机数发生器。 */
static uint32_t next_random(void)
{
    g_rng = (1664525U * g_rng) + 1013904223U;
    return g_rng;
}

/* 返回 [0,1) 均匀随机数。 */
static double random_unit(void)
{
    return (double)(next_random() & 0x00FFFFFFU) / 16777216.0;
}

/* Box-Muller 变换产生一个均值约 0、方差约 1 的高斯随机数。
 * u1 不能为 0，否则 log(0) 无定义，所以设极小下限。 */
static double gaussian_random(void)
{
    double u1 = random_unit();
    double u2 = random_unit();
    if (u1 < 1.0e-12) {
        u1 = 1.0e-12;
    }
    return sqrt(-2.0 * log(u1)) * cos(2.0 * M_PI * u2);
}

/* 单位圆上的最短有符号相位距离，正确处理相位 0/1 的首尾相邻关系。 */
static double cyclic_distance(double phase, double center)
{
    double distance = phase - center;
    if (distance > 0.5) {
        distance -= 1.0;
    } else if (distance < -0.5) {
        distance += 1.0;
    }
    return distance;
}

/* 生成归一化理想形状。方波允许独立上升/下降时间，用斜坡模拟有限带宽；
 * 其余三类忽略 duty 和边沿参数。 */
static double ideal_wave(waveform_type_t type, double phase, double duty,
                         double rise_width, double fall_width)
{
    if (type == WAVEFORM_TYPE_SINE) {
        return sin(2.0 * M_PI * phase);
    }
    if (type == WAVEFORM_TYPE_TRIANGLE) {
        if (phase < 0.25) {
            return 4.0 * phase;
        }
        if (phase < 0.75) {
            return 2.0 - (4.0 * phase);
        }
        return (4.0 * phase) - 4.0;
    }
    if (type == WAVEFORM_TYPE_SAWTOOTH) {
        return 1.0 - (2.0 * phase);
    }
    if (type == WAVEFORM_TYPE_SQUARE) {
        double rise_distance = cyclic_distance(phase, 0.0);
        double fall_distance = cyclic_distance(phase, duty);
        if ((rise_width > 0.0) &&
            (fabs(rise_distance) <= (0.5 * rise_width))) {
            return 2.0 * rise_distance / rise_width;
        }
        if ((fall_width > 0.0) &&
            (fabs(fall_distance) <= (0.5 * fall_width))) {
            return -2.0 * fall_distance / fall_width;
        }
        if ((phase > (0.5 * rise_width)) &&
            (phase < (duty - (0.5 * fall_width)))) {
            return 1.0;
        }
        return -1.0;
    }
    return 0.0;
}

/* 生成“先高斯、再一阶低通”的带限噪声，并把有限记录精确缩放到给定
 * 峰峰值 requested_pp。alpha 来自连续一阶低通的离散指数近似。 */
static void make_lowpass_gaussian_noise(double sample_rate_hz,
                                        double cutoff_hz,
                                        double requested_pp)
{
    double alpha = 1.0 - exp(-2.0 * M_PI * cutoff_hz / sample_rate_hz);
    double state = 0.0;
    double minimum = 1.0e30;
    double maximum = -1.0e30;
    double midpoint;
    double scale;
    uint32_t i;

    /* state += alpha*(input-state) 是最简单的一阶低通递推。 */
    for (i = 0U; i < WAVEFORM_SAMPLE_COUNT; ++i) {
        state += alpha * (gaussian_random() - state);
        g_noise[i] = state;
        if (state < minimum) {
            minimum = state;
        }
        if (state > maximum) {
            maximum = state;
        }
    }
    midpoint = 0.5 * (minimum + maximum);
    scale = (maximum > minimum) ?
            (requested_pp / (maximum - minimum)) : 0.0;
    for (i = 0U; i < WAVEFORM_SAMPLE_COUNT; ++i) {
        g_noise[i] = (g_noise[i] - midpoint) * scale;
    }
}

/* 按本用例自己的采样率生成 4096 点，叠加带限噪声并量化到 12 位 ADC。 */
static void generate_case(uint16_t *samples, waveform_type_t type,
                          double sample_rate_hz, double frequency_hz,
                          double vpp_counts, double center_counts,
                          double duty, double rise_width,
                          double fall_width, double noise_pp_ratio,
                          double noise_bandwidth_ratio, double start_phase)
{
    uint32_t i;

    make_lowpass_gaussian_noise(sample_rate_hz,
                                noise_bandwidth_ratio * frequency_hz,
                                noise_pp_ratio * vpp_counts);
    for (i = 0U; i < WAVEFORM_SAMPLE_COUNT; ++i) {
        double phase = start_phase +
                       (frequency_hz * (double)i / sample_rate_hz);
        double value;
        phase -= floor(phase);
        value = center_counts +
                (0.5 * vpp_counts *
                 ideal_wave(type, phase, duty, rise_width, fall_width)) +
                g_noise[i];
        if (value < 0.0) {
            value = 0.0;
        } else if (value > 4095.0) {
            value = 4095.0;
        }
        samples[i] = (uint16_t)(value + 0.5);
    }
}

/* double 绝对值小工具。 */
static double absolute(double value)
{
    return (value < 0.0) ? -value : value;
}

/* 运行全部高精度随机用例并用进程退出码报告结果。 */
int main(void)
{
    static uint16_t samples[WAVEFORM_SAMPLE_COUNT];
    waveform_type_t type;
    unsigned total = 0U;
    unsigned class_failures = 0U;
    unsigned measurement_failures = 0U;
    double worst_frequency_percent = 0.0;
    double worst_vpp_percent = 0.0;
    double worst_duty_points = 0.0;

    /* 每类 600 例。采样率 18~22 kHz、频率 100~400 Hz、噪声 Vpp 为信号
     * 20%~50%，并覆盖非整数频率和有限方波边沿。 */
    for (type = WAVEFORM_TYPE_SINE; type <= WAVEFORM_TYPE_SAWTOOTH;
         type = (waveform_type_t)(type + 1)) {
        unsigned case_index;
        for (case_index = 0U; case_index < CASES_PER_CLASS; ++case_index) {
            waveform_frame_t frame;
            waveform_result_t result;
            double sample_rate_hz = 18000.0 + (4000.0 * random_unit());
            double frequency_hz = 100.0 + (300.0 * random_unit());
            double vpp_counts = 160.0 + (2800.0 * random_unit());
            double duty = 0.20 + (0.60 * random_unit());
            double rise_width = (type == WAVEFORM_TYPE_SQUARE) ?
                                (0.10 * random_unit()) : 0.0;
            double fall_width = (type == WAVEFORM_TYPE_SQUARE) ?
                                (0.10 * random_unit()) : 0.0;
            double noise_ratio = 0.20 + (0.30 * random_unit());
            double bandwidth_ratio = 1.5 + (1.5 * random_unit());
            double occupied_half_span =
                0.5 * vpp_counts * (1.0 + noise_ratio);
            double center_counts =
                (occupied_half_span + 12.0) +
                ((4071.0 - (2.0 * occupied_half_span)) * random_unit());
            double reference_uv = 3000000.0 + (400000.0 * random_unit());
            double gain_ppm = 800000.0 + (700000.0 * random_unit());
            double expected_vpp_uv;
            double frequency_error;
            double vpp_error;

            if ((case_index % 25U) == 0U) {
                static const double edge_frequencies[] = {
                    100.0, 101.0, 399.0, 400.0
                };
                frequency_hz =
                    edge_frequencies[(case_index / 25U) % 4U];
            }
            if ((type == WAVEFORM_TYPE_SQUARE) &&
                ((case_index % 20U) == 0U)) {
                static const double edge_duties[] = {
                    0.20, 0.21, 0.50, 0.79, 0.80
                };
                duty = edge_duties[(case_index / 20U) % 5U];
                rise_width = 0.10;
                fall_width = 0.10;
            }
            /* 先把 double 采样率量化到接口真实可表达的 0.001 Hz，再用量化
             * 后的值生成数据，保证“真值”和传入元数据完全一致。 */
            frame.sample_rate_millihz =
                (uint32_t)(sample_rate_hz * 1000.0 + 0.5);
            sample_rate_hz =
                (double)frame.sample_rate_millihz / 1000.0;
            generate_case(samples, type, sample_rate_hz, frequency_hz,
                          vpp_counts, center_counts, duty, rise_width,
                          fall_width, noise_ratio, bandwidth_ratio,
                          random_unit());
            frame.samples = samples;
            frame.sample_count = WAVEFORM_SAMPLE_COUNT;
            frame.adc_full_scale = WAVEFORM_ADC_FULL_SCALE;
            frame.adc_reference_microvolts =
                (uint32_t)(reference_uv + 0.5);
            frame.frontend_gain_ppm = (uint32_t)(gain_ppm + 0.5);
            frame.sequence = total + 1U;
            frame.flags = WAVEFORM_FRAME_FLAG_CONTIGUOUS |
                          WAVEFORM_FRAME_FLAG_CLOCK_CALIBRATED |
                          WAVEFORM_FRAME_FLAG_VREF_CALIBRATED;
            ++total;
            if (!waveform_analyze_frame(&frame, &result) ||
                (result.type != type)) {
                ++class_failures;
                if (class_failures <= 16U) {
                    printf("STRONG CLASS FAIL expected=%s got=%s status=%s "
                           "f=%.3f duty=%.3f edge=%.3f/%.3f noise=%.3f "
                           "scores=%u/%u cycles=%u\n",
                           waveform_type_name(type),
                           waveform_type_name(result.type),
                           waveform_status_name(result.status), frequency_hz,
                           duty, rise_width, fall_width, noise_ratio,
                           result.best_score_permille,
                           result.second_score_permille,
                           result.accepted_cycles);
                }
                continue;
            }
            frequency_error =
                absolute(((double)result.frequency_millihz / 1000.0) -
                         frequency_hz) / frequency_hz * 100.0;
            /* 源端期望 Vpp = ADC 码幅度/满量程*参考电压/前端增益。增益以
             * ppm 表示，因此公式额外乘 1,000,000。 */
            expected_vpp_uv =
                vpp_counts * reference_uv /
                (double)WAVEFORM_ADC_FULL_SCALE *
                1000000.0 / gain_ppm;
            vpp_error =
                absolute((double)result.vpp_microvolts - expected_vpp_uv) /
                expected_vpp_uv * 100.0;
            if (frequency_error > worst_frequency_percent) {
                worst_frequency_percent = frequency_error;
            }
            if (vpp_error > worst_vpp_percent) {
                worst_vpp_percent = vpp_error;
            }
            if ((frequency_error > 0.35) || (vpp_error > 9.0)) {
                ++measurement_failures;
            }
            if (type == WAVEFORM_TYPE_SQUARE) {
                double measured_duty =
                    (double)result.duty_ppm / 1000000.0;
                double duty_error =
                    absolute(measured_duty - duty) * 100.0;
                if (duty_error > worst_duty_points) {
                    worst_duty_points = duty_error;
                }
                if (duty_error > 2.5) {
                    ++measurement_failures;
                    if (measurement_failures <= 16U) {
                        printf("STRONG DUTY FAIL f=%.3f expected=%.4f "
                               "got=%.4f edge=%.3f/%.3f noise=%.3f\n",
                               frequency_hz, duty, measured_duty,
                               rise_width, fall_width, noise_ratio);
                    }
                }
            }
        }
    }
    printf("strong_cases=%u class_fail=%u measurement_fail=%u "
           "worst_freq=%.4f%% worst_vpp=%.3f%% "
           "worst_duty=%.3f percentage-point\n",
           total, class_failures, measurement_failures,
           worst_frequency_percent, worst_vpp_percent, worst_duty_points);
    /* 0 表示全部满足分类及误差门限。 */
    return ((class_failures == 0U) &&
            (measurement_failures == 0U)) ? 0 : 1;
}
